cmake_minimum_required(VERSION 3.8)
project(k1muse_web)
find_package(ament_cmake REQUIRED)
find_package(nlohmann_json REQUIRED)
set(deps rclcpp rclcpp_action rcl_interfaces ament_index_cpp geometry_msgs nav_msgs sensor_msgs
  std_msgs std_srvs tf2_ros nav2_msgs lifecycle_msgs k1muse_interfaces)
foreach(dep IN LISTS deps)
  find_package(${dep} REQUIRED)
endforeach()
add_executable(web_operator src/web_operator.cpp)
target_compile_features(web_operator PRIVATE cxx_std_17)
target_compile_options(web_operator PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(web_operator PRIVATE include)
target_link_libraries(web_operator nlohmann_json::nlohmann_json)
ament_target_dependencies(web_operator ${deps})
install(TARGETS web_operator DESTINATION lib/${PROJECT_NAME})
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/web/index.html")
  message(FATAL_ERROR "Build apps/k1muse_web with npm run build before colcon/k1 build")
endif()
install(DIRECTORY launch config scripts web DESTINATION share/${PROJECT_NAME}
  PATTERN "__pycache__" EXCLUDE)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_control test/test_control.cpp)
  target_include_directories(test_control PRIVATE include)
  target_compile_features(test_control PRIVATE cxx_std_17)
  ament_add_gtest(test_child test/test_child.cpp)
  target_include_directories(test_child PRIVATE include)
  target_compile_features(test_child PRIVATE cxx_std_17)
  ament_add_gtest(test_intent_web test/test_intent_web.cpp)
  target_compile_features(test_intent_web PRIVATE cxx_std_17)
  target_include_directories(test_intent_web PRIVATE include)
  target_link_libraries(test_intent_web nlohmann_json::nlohmann_json)
  ament_target_dependencies(test_intent_web ${deps})
  ament_add_gtest(test_navigation_web test/test_navigation_web.cpp)
  target_compile_features(test_navigation_web PRIVATE cxx_std_17)
  target_include_directories(test_navigation_web PRIVATE include)
  target_link_libraries(test_navigation_web nlohmann_json::nlohmann_json)
  ament_target_dependencies(test_navigation_web ${deps})
  ament_add_gtest(test_warehouse_web test/test_warehouse_web.cpp TIMEOUT 30)
  target_compile_features(test_warehouse_web PRIVATE cxx_std_17)
  target_include_directories(test_warehouse_web PRIVATE include)
  target_link_libraries(test_warehouse_web nlohmann_json::nlohmann_json)
  target_compile_definitions(test_warehouse_web PRIVATE
    K1MUSE_WEB_HELPER="${CMAKE_CURRENT_SOURCE_DIR}/scripts/map_helper.py"
    K1MUSE_WAREHOUSE_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/../k1muse_manager/test/data/warehouse_cases.json")
  ament_target_dependencies(test_warehouse_web ${deps})
endif()
ament_package()
